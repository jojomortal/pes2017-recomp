cmake_minimum_required(VERSION 3.20)
project(pes2017recomp CXX)

# ARMv8.2-A baseline for the phone build: without it the NDK emits calls to the
# outline-atomics helpers for every guest lwarx/stwcx. (std::atomic CAS) instead
# of one LSE instruction. Every phone since the Snapdragon 845 / Cortex-A75 has it;
# the app checks /proc/cpuinfo before loading (NativeBridge.kt). Turn off for a
# build that must run on ARMv8.0 cores.
option(PES_ARMV82 "Build for ARMv8.2-A (LSE atomics, fp16, dotprod) on Android" ON)
if(ANDROID AND PES_ARMV82)
    string(APPEND CMAKE_C_FLAGS " -march=armv8.2-a+fp16+dotprod")
    string(APPEND CMAKE_CXX_FLAGS " -march=armv8.2-a+fp16+dotprod")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# For the checks that assert rather than report, so `ctest` is a yes or a no.
# Everything registered runs without game files, a disc or a GPU.
enable_testing()
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Android brings its own window, input and audio (runtime/android/), so SDL is
# off there whatever the cache says -- and a default the Gradle build does not
# have to repeat.
if(ANDROID)
    set(PES17_USE_SDL OFF CACHE BOOL "Use SDL3 (not on Android)" FORCE)
endif()
option(PES17_USE_SDL "Use SDL3 for gamepad input (and later window/audio)" ON)
option(PES17_TRACE_INDIRECT "Report indirect calls with no recompiled target instead of jumping to null" ON)
option(PES17_LOGGING "Emit the runtime's log" ON)
option(PES17_PORTABLE "Link the C++ runtime statically, for a binary that runs on other machines" OFF)
option(PES17_DIAGNOSTICS "Build the traces, dumps, statistics, reports and test harness (runtime/diagnostics.h)" ON)

# ---- which executable ------------------------------------------------------
# PES 2017 ships one executable, default.xex; build.sh recompiles it into ppc/.
set(PES17_TITLE "sp")
set(PES17_VERSION "tu0")
set(PES17_PPC_DIR "${CMAKE_SOURCE_DIR}/ppc")

file(GLOB PPC_SOURCES CONFIGURE_DEPENDS "${PES17_PPC_DIR}/ppc_recomp.*.cpp")
list(APPEND PPC_SOURCES "${PES17_PPC_DIR}/ppc_func_mapping.cpp")

# Forçar a biblioteca de texto iconv nativa no ecossistema da Apple (iOS)
if(APPLE)
    link_libraries(iconv)
endif()

if(NOT PPC_SOURCES)
    message(FATAL_ERROR "${PES17_PPC_DIR} is empty -- run TITLE=${PES17_TITLE} VERSION=${PES17_VERSION} ./build.sh first to generate it")
endif()

add_library(ppc_recomp STATIC ${PPC_SOURCES})
set(SIMDE_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/third_party/XenonRecomp/thirdparty/simde")
if(NOT EXISTS "${SIMDE_INCLUDE_DIR}/simde/simde-common.h" AND EXISTS "${CMAKE_SOURCE_DIR}/third_party/simde/simde/simde-common.h")
    set(SIMDE_INCLUDE_DIR "${CMAKE_SOURCE_DIR}/third_party/simde")
endif()
target_include_directories(ppc_recomp PUBLIC
    "${PES17_PPC_DIR}"
    "${SIMDE_INCLUDE_DIR}"
    "${SIMDE_INCLUDE_DIR}/simde")
# The generated code is machine output; its warnings are noise.
target_compile_options(ppc_recomp PRIVATE -w)
# PowerPC separate fmul/fadd must stay separate (fused forms are emitted
# explicitly where the guest fuses), and the generated accesses alias through
# the guest base pointer: contraction and strict aliasing would change results.
target_compile_options(ppc_recomp PRIVATE -ffp-contract=off -fno-strict-aliasing)
# The console's hardware registers are plain memory in this address space. The
# recompiler marks the stores that address them (eieio follows), and this header
# routes those through a hook so the XMA decoder sees its kick, lock and clear
# registers written. Force-included: ppc/ is regenerated by build.sh.
target_compile_options(ppc_recomp PRIVATE -include "${CMAKE_SOURCE_DIR}/runtime/mmio_hook.h")
option(PES_TRAP_INF "Report stores of infinity (arm with files/trap_inf); slows every store" OFF)
if(PES_TRAP_INF)
    target_compile_definitions(ppc_recomp PRIVATE PES_TRAP_INF=1)
endif()
if(NOT PES17_LOGGING)
    # Compiles every LOGI/LOGW/LOGE/LOGK call to nothing, warnings included.
    add_compile_definitions(PES17_NO_LOGGING=1)
endif()

if(PES17_TRACE_INDIRECT)
    target_compile_definitions(ppc_recomp PUBLIC PES17_TRACE_INDIRECT=1)
endif()

# ---- runtime ---------------------------------------------------------------
file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS
     "${CMAKE_SOURCE_DIR}/runtime/*.cpp"
     "${CMAKE_SOURCE_DIR}/runtime/kernel/*.cpp"
     "${CMAKE_SOURCE_DIR}/runtime/gpu/*.cpp"
     "${CMAKE_SOURCE_DIR}/runtime/apu/*.cpp"
     "${CMAKE_SOURCE_DIR}/runtime/install/*.cpp"
     "${CMAKE_SOURCE_DIR}/runtime/gpu/vulkan/*.cpp"
     "${CMAKE_SOURCE_DIR}/runtime/gpu/xe/*.cpp")

# Xenia's shader interpreter, vendored with shims for the rex/ headers it
# includes; runs memory-export vertex shaders on the CPU (runtime/gpu/xe).
set_source_files_properties(
    "${CMAKE_SOURCE_DIR}/runtime/gpu/xe/interpreter.cpp"
    "${CMAKE_SOURCE_DIR}/runtime/gpu/xe/memexport.cpp"
    "${CMAKE_SOURCE_DIR}/runtime/gpu/xe/ucode.cpp"
    PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_SOURCE_DIR}/runtime/gpu/xe/shim")

if(ANDROID)
    # The window, the touch pad, the AAudio sink, the driver loader and the
    # JNI surface the app talks to (docs/android.md).
    file(GLOB ANDROID_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/runtime/android/*.cpp")
    list(APPEND RUNTIME_SOURCES ${ANDROID_SOURCES} "${CMAKE_SOURCE_DIR}/launcher/disc.cpp")
    # The title-update package reader (dependency-free STFS): the TU6 apps
    # apply the update on the device after copying the disc.
    list(APPEND RUNTIME_SOURCES "${CMAKE_SOURCE_DIR}/launcher/package.cpp")
    # XenonRecomp's patcher, which turns the disc's executables into the
    # update's. Vendored in third_party/XenonRecomp, so present in every checkout;
    # kept conditional in case the tree is trimmed, where TU6 apps report that the
    # APK must come from CI instead of failing to link.
    set(XENON "${CMAKE_SOURCE_DIR}/third_party/XenonRecomp")
    if(EXISTS "${XENON}/XenonUtils/xex_patcher.cpp" AND EXISTS "${XENON}/XenonUtils/xex_patcher.h")
        enable_language(C)
        list(APPEND RUNTIME_SOURCES
            "${XENON}/XenonUtils/xex_patcher.cpp"
            "${XENON}/XenonUtils/memory_mapped_file.cpp"
            "${XENON}/thirdparty/libmspack/libmspack/mspack/lzxd.c"
            "${XENON}/thirdparty/tiny-AES-c/aes.c")
        set(PES17_XENON_INCLUDES
            "${XENON}/XenonUtils"
            "${XENON}/thirdparty/simde"
            "${XENON}/thirdparty/libmspack/libmspack/mspack"
            "${XENON}/thirdparty/tiny-AES-c"
            "${XENON}/thirdparty/TinySHA1")
    endif()
endif()

# ---- online service --------------------------------------------------------
set(PES17_ONLINE "none" CACHE STRING "Online service: none, lan or steam")
if(PES17_ONLINE STREQUAL "none")
    set(PES17_ONLINE_BACKENDS "")
elseif(PES17_ONLINE STREQUAL "lan")
    set(PES17_ONLINE_BACKENDS lan)
elseif(PES17_ONLINE STREQUAL "steam")
    set(PES17_ONLINE_BACKENDS steam lan)
else()
    message(FATAL_ERROR "PES17_ONLINE=${PES17_ONLINE}: none, lan or steam")
endif()
set(PES17_ONLINE_SOURCES "")
set(PES17_ONLINE_DEFINITIONS "")
foreach(backend ${PES17_ONLINE_BACKENDS})
    list(APPEND PES17_ONLINE_SOURCES "${CMAKE_SOURCE_DIR}/runtime/online/${backend}.cpp")
    string(TOUPPER "${backend}" backend_upper)
    list(APPEND PES17_ONLINE_DEFINITIONS "PES17_ONLINE_${backend_upper}=1")
endforeach()
message(STATUS "Online service: ${PES17_ONLINE}")

if(ANDROID)
    add_library(pes17 SHARED ${RUNTIME_SOURCES} "${CMAKE_SOURCE_DIR}/runtime/online/service.cpp" ${PES17_ONLINE_SOURCES})
    target_compile_definitions(pes17 PRIVATE ${PES17_ONLINE_DEFINITIONS})
    target_link_libraries(pes17 PRIVATE ppc_recomp log android EGL GLESv3 OpenSLES aaudio)
else()
    add_executable(pes17 ${RUNTIME_SOURCES} "${CMAKE_SOURCE_DIR}/runtime/online/service.cpp" ${PES17_ONLINE_SOURCES})
    target_compile_definitions(pes17 PRIVATE ${PES17_ONLINE_DEFINITIONS})
    
    # Vinculando apenas o núcleo do recompilador e frameworks da própria Apple de forma limpa
    target_link_libraries(pes17 PRIVATE 
        ppc_recomp 
        "-framework AudioToolbox" 
        "-framework VideoToolbox" 
        "-framework CoreMedia"
        "-framework CoreVideo"
    )
    
    # Incluir o caminho do runtime para que as subpastas localizem o arquivo title.h
    target_include_directories(pes17 PRIVATE "${CMAKE_SOURCE_DIR}/runtime")
endif()
